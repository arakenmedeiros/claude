@@EX 1 | Consenso: qual ligação e quantas partições? | básico
Em dois conjuntos não convexos (**luas** e **anéis**), gere ensembles de 5, 20 e 50 partições de K-means (com *k* entre 8 e 15, usando 80 % dos pontos), calcule a matriz de coassociação e extraia o consenso com as ligações **simples**, **média** e **completa**. Monte uma tabela de ARI e explique por que a ligação completa falha.
@@SOL
```python
import numpy as np
from sklearn.datasets import make_moons, make_circles
from sklearn.cluster import KMeans, AgglomerativeClustering
from sklearn.metrics import adjusted_rand_score

def consenso(X, n_particoes, ligacao, k_final=2, seed=0):
    rng = np.random.default_rng(seed)
    C = np.zeros((len(X), len(X)))
    for _ in range(n_particoes):
        k = int(rng.integers(8, 16))
        idx = rng.choice(len(X), int(0.8 * len(X)), replace=False)
        p = KMeans(k, n_init=1, random_state=int(rng.integers(10**6))).fit(X[idx]).predict(X)
        C += p[:, None] == p[None, :]
    C /= n_particoes
    return AgglomerativeClustering(k_final, metric="precomputed", linkage=ligacao).fit_predict(1 - C)

conjuntos = {
    "Luas":   make_moons(400, noise=0.08, random_state=0),
    "Anéis":  make_circles(400, factor=0.4, noise=0.05, random_state=0),
}
print(f"{'Conjunto':8s} {'partições':>9s} {'simples':>8s} {'média':>8s} {'completa':>9s}")
for nome, (X, y) in conjuntos.items():
    for m in (5, 20, 50):
        aris = [adjusted_rand_score(y, consenso(X, m, lig)) for lig in ("single", "average", "complete")]
        print(f"{nome:8s} {m:9d} " + " ".join(f"{a:8.2f}" for a in aris))
```
@@NOTE A ligação média é a mais estável (ARI de 0,98 a 1,00 nas luas, mesmo com 5 partições) e a simples melhora com mais partições (0,70 com 5 e 1,00 com 20). A completa fracassa nas luas (ARI de 0,05): ela exige grupos compactos e, em formas alongadas, junta pedaços de luas diferentes. Nos anéis, as ligações simples e média acertam, e a completa também falha.
@@EX 2 | Quantas restrições bastam? | intermediário
Use o COP-K-means (Seção 13.3) em três blobs sobrepostos. Sorteie 0, 10, 30, 60 e 100 restrições (a partir dos rótulos verdadeiros, simulando um especialista), com 10 sorteios cada, e calcule o ARI médio, as violações médias e quantas execuções foram viáveis.
@@SOL
```python
import numpy as np
from sklearn.datasets import make_blobs
from sklearn.cluster import KMeans
from sklearn.metrics import adjusted_rand_score

X, y = make_blobs(300, centers=3, cluster_std=2.4, random_state=3)

def sortear_restricoes(n_pares, seed):
    rng = np.random.default_rng(seed)
    ml, cl = [], []
    while len(ml) + len(cl) < n_pares:
        i, j = rng.choice(len(X), 2, replace=False)
        (ml if y[i] == y[j] else cl).append((int(i), int(j)))
    return ml, cl

def cop_kmeans(X, k, ml, cl, n_iter=50, seed=0):
    r = np.random.default_rng(seed)
    pai = list(range(len(X)))
    def raiz(a):
        while pai[a] != a:
            pai[a] = pai[pai[a]]; a = pai[a]
        return a
    for a, b in ml: pai[raiz(a)] = raiz(b)
    comp = np.array([raiz(i) for i in range(len(X))])
    proib = {}
    for a, b in cl:
        proib.setdefault(a, set()).add(b); proib.setdefault(b, set()).add(a)
    centros = X[r.choice(len(X), k, replace=False)]
    rot = -np.ones(len(X), int)
    for _ in range(n_iter):
        novo = -np.ones(len(X), int)
        for i in r.permutation(len(X)):
            if novo[i] >= 0: continue
            membros = np.where(comp == comp[i])[0]
            d = ((X[membros][:, None] - centros[None]) ** 2).sum(2).sum(0)
            for c in np.argsort(d):
                if not any(novo[v] == c for m in membros for v in proib.get(m, ())):
                    novo[membros] = c; break
            else:
                return None
        if (novo == rot).all(): break
        rot = novo
        centros = np.array([X[rot == c].mean(0) if (rot == c).any() else centros[c] for c in range(k)])
    return rot

def viol(rot, ml, cl):
    return sum(rot[a] != rot[b] for a, b in ml) + sum(rot[a] == rot[b] for a, b in cl)

print("n restrições | ARI médio | violações médias | execuções viáveis (de 10 sorteios)")
for n in (0, 10, 30, 60, 100):
    aris, vs = [], []
    for s in range(10):
        ml, cl = sortear_restricoes(n, seed=s)
        melhor = None
        for ini in range(5):
            r = cop_kmeans(X, 3, ml, cl, seed=ini)
            if r is not None:
                sse = sum(((X[r == c] - X[r == c].mean(0)) ** 2).sum() for c in range(3))
                if melhor is None or sse < melhor[0]: melhor = (sse, r)
        if melhor is not None:
            aris.append(adjusted_rand_score(y, melhor[1])); vs.append(viol(melhor[1], ml, cl))
    print(f"{n:12d} | {np.mean(aris):9.2f} | {np.mean(vs):16.1f} | {len(aris):2d}")
```
@@NOTE As restrições garantem violações zero, mas o ganho em ARI é pequeno (de 0,71 para 0,76 com 100 restrições): os grupos se sobrepõem tanto que poucos pares ajudam. Com 100 restrições, uma das 10 execuções já fica inviável: o algoritmo guloso pode não achar solução. Restrições valem mais quando são **informativas** (pares na fronteira) do que quando são sorteadas ao acaso.
@@EX 3 | Dados mistos: o peso das categorias na distância de Gower | intermediário
Gere clientes com idade, renda, plano (ordinal) e canal (categórico). Implemente a distância de Gower com um peso `w_cat` para o atributo categórico, agrupe com ligação média e varie `w_cat` em 0; 0,5; 1; 2; 4 e 8. Calcule o ARI com os perfis ocultos e discuta.
@@SOL
```python
import numpy as np
import pandas as pd
from sklearn.cluster import AgglomerativeClustering
from sklearn.metrics import adjusted_rand_score

rng = np.random.default_rng(0)
n = 450
perfil = rng.integers(0, 3, n)
canais = [["app", "whatsapp"], ["loja", "telefone"], ["site", "email"]]
todos = ["app", "whatsapp", "loja", "telefone", "site", "email"]
canal = [rng.choice(canais[p]) if rng.random() < 0.8 else rng.choice(todos) for p in perfil]
pr = np.array([[0.6, 0.3, 0.1], [0.1, 0.3, 0.6], [0.2, 0.6, 0.2]])[perfil]
plano = [rng.choice([0, 1, 2], p=p) for p in pr]
df = pd.DataFrame({"idade": rng.normal(np.array([32, 42, 52])[perfil], 11),
                   "renda": rng.normal(np.array([3.5, 5.0, 4.0])[perfil], 1.6),
                   "plano": plano, "canal": canal})

def gower(df, numericos, categoricos, w_cat=1.0):
    D = np.zeros((len(df), len(df))); soma_pesos = 0.0
    for c in numericos:
        v = df[c].to_numpy(float); D += np.abs(v[:, None] - v[None, :]) / (v.max() - v.min()); soma_pesos += 1
    for c in categoricos:
        v = df[c].to_numpy(); D += w_cat * (v[:, None] != v[None, :]); soma_pesos += w_cat
    return D / soma_pesos

print("peso da categoria 'canal' | ARI (Gower + ligação média)")
for w in (0.0, 0.5, 1.0, 2.0, 4.0, 8.0):
    G = gower(df, ["idade", "renda", "plano"], ["canal"], w_cat=w)
    r = AgglomerativeClustering(3, metric="precomputed", linkage="average").fit_predict(G)
    print(f"{w:25.1f} | {adjusted_rand_score(perfil, r):.2f}")
```
@@NOTE Ignorar o canal (peso 0) dá ARI de apenas 0,18, porque os atributos numéricos se sobrepõem muito. Com peso 0,5 o ARI sobe a 0,41 e, a partir de 1, estabiliza em 0,67. Em dados reais, o peso adequado depende do problema e deve ser validado (estabilidade, interpretação), não ajustado ao gabarito.
@@EX 4 | Biclustering: até que ruído a estrutura em blocos sobrevive? | intermediário
Com `make_biclusters` (300 × 300, 4 biclusters), embaralhe linhas e colunas, aplique `SpectralCoclustering` e calcule o `consensus_score` para níveis de ruído de 5, 50, 100, 150 e 250.
@@SOL
```python
import numpy as np
from sklearn.datasets import make_biclusters
from sklearn.cluster import SpectralCoclustering
from sklearn.metrics import consensus_score

print("ruído | consensus_score")
for ruido in (5, 50, 100, 150, 250):
    dados, linhas, colunas = make_biclusters(shape=(300, 300), n_clusters=4, noise=ruido, shuffle=False, random_state=0)
    rng = np.random.default_rng(0)
    pl, pc = rng.permutation(300), rng.permutation(300)
    emb = dados[pl][:, pc]
    cc = SpectralCoclustering(n_clusters=4, random_state=0).fit(emb)
    verdade = (linhas[:, pl], colunas[:, pc])
    print(f"{ruido:5d} | {consensus_score(cc.biclusters_, verdade):.2f}")
```
@@NOTE O escore de consenso é 1,00 até o ruído 50, cai para 0,89 com ruído 100, 0,60 com 150 e 0,13 com 250: a partir de certo ponto, o ruído esconde os blocos. Em dados reais, o nível de ruído não é conhecido, e a validação deve combinar o escore interno, a estabilidade e a leitura de especialistas.
@@EX 5 | Um alarme de deriva por taxa de rejeição | avançado
Treine um K-means em um lote de referência. Defina o limite de rejeição como o percentil 99 das distâncias ao centro no treino. Simule 13 lotes, com os centros deslocados a partir do lote 8, e calcule em cada lote a taxa de rejeitados e a distância média ao centro. Dispare um alerta quando a taxa passar de 5 %.
@@SOL
```python
import numpy as np
from sklearn.cluster import KMeans

rng = np.random.default_rng(1)
base = np.array([[0.0, 0.0], [6.0, 0.0], [3.0, 5.0]])

def lote(t, n=300):
    desloc = np.zeros_like(base)
    if t >= 8:
        desloc[:, 0] += 1.5 * (t - 7)                   # deriva abrupta e crescente a partir do lote 8
    y = rng.integers(0, 3, n)
    return (base + desloc)[y] + rng.normal(0, 0.8, (n, 2))

ref = lote(0)
modelo = KMeans(3, n_init=10, random_state=0).fit(ref)
d_ref = modelo.transform(ref).min(1)
limite = np.quantile(d_ref, 0.99)                       # limite de rejeição: percentil 99 do treino

print(" t | % rejeitados | distância média ao centro | alerta (rejeição > 5 %)")
for t in range(1, 14):
    X = lote(t)
    d = modelo.transform(X).min(1)
    taxa = (d > limite).mean()
    print(f"{t:2d} | {100 * taxa:12.1f} | {d.mean():25.2f} | {'ALERTA' if taxa > 0.05 else '-'}")
```
@@NOTE Nos lotes 1 a 7, a taxa de rejeição fica abaixo de 4 %, como esperado do percentil 99. No lote 8, em que a deriva começa, ela salta para cerca de 21 % e continua subindo, e a distância média ao centro dispara: o alerta funciona. Um limite de 5 % é uma escolha de negócio: reduza-o para detectar antes (com mais falsos alarmes) ou aumente-o para alarmar menos.
@@PROJ Segmentação de clientes em produção: consenso, dados mistos e monitoramento | Cap13
**Contexto.** Uma seguradora (fictícia) quer segmentar os clientes por atributos **mistos** (idade, renda, plano e canal de contratação) e **usar** a segmentação nos meses seguintes, sem reajustá-la a cada semana. A equipe precisa de um agrupamento **robusto**, de uma regra para **atribuir** clientes novos e de um **alarme** para quando surgir um perfil que o modelo não conhece.

**Fonte dos dados.** Dados simulados (função `gerar` na solução), com três perfis no treino e um quarto perfil que surge dois meses depois.

**Tarefas.**
1. Gere 600 clientes de treino (3 perfis), 300 clientes do mês 1 (mesmos perfis) e 300 do mês 2 (com o perfil novo). **Ajuste todo o pré-processamento apenas no treino.**
2. Construa um **ensemble de 30 partições** de K-means (*k* de 3 a 8) sobre os dados codificados (padronização + *one-hot*), calcule a coassociação e extraia o **consenso** com ligação média.
3. Escolha *k* com a **força de predição** (divisão interna do treino) e justifique.
4. Atribua os clientes novos com um classificador de *k* vizinhos treinado nos rótulos do consenso e **rejeite** os que estão longe demais (limite: percentil 99 das distâncias do treino).
5. Calcule a taxa de rejeição nos meses 1 e 2 e discuta o alarme de deriva.
6. Apenas no final, use os perfis ocultos para calcular o ARI do consenso e o dos casos aceitos.
7. Escreva um parágrafo de recomendação: o que a equipe deve monitorar e quando reagrupar.

**Entregáveis.** Script, tabela de resultados e parágrafo.
@@SOL
```python
import numpy as np
import pandas as pd
from sklearn.cluster import KMeans, AgglomerativeClustering
from sklearn.compose import ColumnTransformer
from sklearn.neighbors import KNeighborsClassifier, NearestNeighbors
from sklearn.preprocessing import OneHotEncoder, StandardScaler
from sklearn.metrics import adjusted_rand_score

# ---------- 1) Dados sintéticos: clientes com atributos mistos
CANAIS = [["app", "whatsapp"], ["loja", "telefone"], ["site", "email"], ["parceiro", "evento"]]
TODOS = sum(CANAIS, [])
IDADE, RENDA = np.array([30, 42, 52, 36]), np.array([3.5, 5.0, 4.0, 7.0])
PLANO = np.array([[0.6, 0.3, 0.1], [0.1, 0.3, 0.6], [0.2, 0.6, 0.2], [0.1, 0.2, 0.7]])

def gerar(n, perfis, seed):
    rng = np.random.default_rng(seed)
    p = rng.choice(perfis, n)
    canal = [rng.choice(CANAIS[k]) if rng.random() < 0.8 else rng.choice(TODOS) for k in p]
    df = pd.DataFrame({"idade": rng.normal(IDADE[p], 8), "renda": rng.normal(RENDA[p], 1.2),
                       "plano": [rng.choice(["basico", "padrao", "premium"], p=PLANO[k]) for k in p],
                       "canal": canal})
    return df, p

treino, y_tr = gerar(600, [0, 1, 2], seed=0)
mes1, y_m1 = gerar(300, [0, 1, 2], seed=1)           # mês seguinte: mesmos perfis
mes2, y_m2 = gerar(300, [0, 1, 2, 3], seed=2)        # dois meses depois: surge o perfil 3

# ---------- 2) Pré-processamento ajustado SÓ no treino
prep = ColumnTransformer([("num", StandardScaler(), ["idade", "renda"]),
                          ("cat", OneHotEncoder(handle_unknown="ignore", sparse_output=False), ["plano", "canal"])]).fit(treino)
Z_tr, Z_m1, Z_m2 = prep.transform(treino), prep.transform(mes1), prep.transform(mes2)

# ---------- 3) Consenso: 30 partições K-means (k de 3 a 8) -> coassociação -> hierárquico (média)
rng = np.random.default_rng(0)
C = np.zeros((len(Z_tr), len(Z_tr))); M = 30
for _ in range(M):
    k = int(rng.integers(3, 9)); idx = rng.choice(len(Z_tr), int(0.8 * len(Z_tr)), replace=False)
    p = KMeans(k, n_init=1, random_state=int(rng.integers(10**6))).fit(Z_tr[idx]).predict(Z_tr)
    C += p[:, None] == p[None, :]
C /= M

# ---------- 4) Força de predição para escolher k (divisão treino/validação do próprio treino)
def forca(k):
    a, b = Z_tr[:300], Z_tr[300:]
    ma, mb = KMeans(k, n_init=10, random_state=0).fit(a), KMeans(k, n_init=10, random_state=0).fit(b)
    prev, ps = ma.predict(b), []
    for j in range(k):
        i = np.where(mb.labels_ == j)[0]
        ps.append(((prev[i][:, None] == prev[i][None, :]).sum() - len(i)) / max(len(i) * (len(i) - 1), 1))
    return min(ps)
for k in range(2, 7):
    print(f"Força de predição k = {k}: {forca(k):.2f}")

k_final = 3
rotulos = AgglomerativeClustering(k_final, metric="precomputed", linkage="average").fit_predict(1 - C)
print("Tamanho dos grupos do consenso:", np.bincount(rotulos))

# ---------- 5) Atribuir casos novos (kNN) e rejeitar os distantes demais
knn = KNeighborsClassifier(15).fit(Z_tr, rotulos)
d5 = NearestNeighbors(n_neighbors=6).fit(Z_tr).kneighbors(Z_tr)[0][:, 1:].mean(1)   # distância média a 5 vizinhos (treino)
limite = np.quantile(d5, 0.99)

def monitorar(nome, Z, y_real):
    d = NearestNeighbors(n_neighbors=5).fit(Z_tr).kneighbors(Z)[0].mean(1)
    rej = d > limite
    atrib = knn.predict(Z)
    ok = ~rej
    print(f"{nome}: rejeitados = {100 * rej.mean():4.1f} % | ARI (aceitos, rótulos ocultos) = {adjusted_rand_score(y_real[ok], atrib[ok]):.2f}")
    return rej

print("ARI do consenso no treino (rótulos ocultos):", round(adjusted_rand_score(y_tr, rotulos), 2))
r1 = monitorar("Mês 1 (sem mudança)  ", Z_m1, y_m1)
r2 = monitorar("Mês 2 (perfil novo)  ", Z_m2, y_m2)
print("Entre os clientes do perfil novo, rejeitados:", f"{100 * r2[y_m2 == 3].mean():.1f} %")
print("Entre os clientes dos perfis antigos, rejeitados:", f"{100 * r2[y_m2 != 3].mean():.1f} %")
```
@@RUBRICA
- Pré-processamento ajustado só no treino, sem vazamento.
- Consenso construído com partições diversas e extraído com ligação adequada.
- Escolha de k justificada pela força de predição (e não só pelo gabarito).
- Atribuição de casos novos com regra de rejeição explícita.
- Taxa de rejeição comparada entre os meses e interpretada como sinal de deriva.
- Rótulos ocultos usados apenas no final.
- Recomendação clara, com limites e responsabilidades humanas.
@@NOTE A força de predição é alta para k = 2 (0,89) e cai a 0,77 em k = 3; o consenso do treino tem ARI de cerca de 0,45 com os perfis ocultos, o que mostra como a sobreposição dificulta o problema. No mês 1, a taxa de rejeição fica em cerca de 2 %; no mês 2, sobe a cerca de 13 %, e entre os clientes do perfil novo, a 41 %, contra cerca de 3 % nos perfis antigos. O alarme de deriva funciona, mas o perfil novo só é plenamente identificável depois de reagrupar com dados atualizados.
