@@EX 1 | O efeito de C e gamma com poucos rótulos | básico

*Pergunta:* a escolha de C e de gamma importa com tão poucos rótulos?

from sklearn.svm import SVC
from sklearn.datasets import make_moons

@@EX 2 | TSVM no Breast Cancer | intermediário
No **Breast Cancer** (UCI), com atributos padronizados, 10 rótulos (5 por classe) e 20 sorteios, compare a `SVC` (RBF, `gamma="scale"`, C = 10) só com os rotulados e o TSVM simplificado do capítulo (mesmos parâmetros, Cu final = 1). Para o TSVM, use `gamma` igual ao que o modo "scale" calcularia: `1 / (n_atributos * variância dos dados)`. Rode o TSVM de duas formas: com a proporção de positivos estimada nos **rotulados** (a padrão) e com a proporção **real** do treino (`prop_pos=y_tr.mean()`).

*Pergunta:* o TSVM ajudou neste conjunto? O que a comparação das duas proporções revela?
@@SOL
```python
import numpy as np
from sklearn.svm import SVC
from sklearn.datasets import load_breast_cancer
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler

def tsvm(X, y_parcial, C=10.0, gamma=1.0, Cu_final=1.0, passos=5, max_trocas=40, prop_pos=None, balancear=True):
    """TSVM simplificado, inspirado em Joachims (1999). Rótulos 0/1; -1 = sem rótulo.
    1) treina uma SVM só com os rotulados e dá rótulos provisórios aos demais, respeitando a proporção de positivos;
    2) aumenta aos poucos o peso Cu dos exemplos sem rótulo e, a cada passo, troca o rótulo de pares que violam a margem."""
    rot = np.where(y_parcial != -1)[0]; sem = np.where(y_parcial == -1)[0]
    Xl, yl = X[rot], 2 * y_parcial[rot] - 1; Xu = X[sem]
    if prop_pos is None: prop_pos = np.mean(yl == 1)
    m = SVC(C=C, kernel="rbf", gamma=gamma).fit(Xl, yl)
    f = m.decision_function(Xu)
    if balancear:
        yu = -np.ones(len(sem)); yu[np.argsort(-f)[:int(round(prop_pos * len(sem)))]] = 1      # proporção fixa de positivos
    else:
        yu = np.where(f > 0, 1.0, -1.0)
    Xa = np.vstack([Xl, Xu])
    for Cu in np.geomspace(Cu_final / 100, Cu_final, passos):                                   # peso crescente dos sem rótulo
        for _ in range(max_trocas):
            sw = np.r_[np.ones(len(rot)), np.full(len(sem), Cu)]
            m = SVC(C=C, kernel="rbf", gamma=gamma).fit(Xa, np.r_[yl, yu], sample_weight=sw)
            folga = np.maximum(0, 1 - yu * m.decision_function(Xu))                              # quanto cada exemplo viola a margem
            pos = np.where((yu == 1) & (folga > 0))[0]; neg = np.where((yu == -1) & (folga > 0))[0]
            if len(pos) == 0 or len(neg) == 0: break
            i, j = pos[np.argmax(folga[pos])], neg[np.argmax(folga[neg])]
            if folga[i] + folga[j] <= 2: break                                                   # trocar não compensa
            yu[i], yu[j] = -1, 1                                                                 # troca os rótulos do par
    return m

X, y = load_breast_cancer(return_X_y=True)
X_tr, X_te, y_tr, y_te = train_test_split(X, y, test_size=0.3, stratify=y, random_state=0)
esc = StandardScaler().fit(X_tr); X_tr, X_te = esc.transform(X_tr), esc.transform(X_te)
gamma = 1 / (X_tr.shape[1] * X_tr.var())
a, b, c = [], [], []
for s in range(20):
    r = np.random.RandomState(s); idx = np.r_[r.choice(np.where(y_tr == 0)[0], 5, replace=False), r.choice(np.where(y_tr == 1)[0], 5, replace=False)]
    yp = np.full(len(y_tr), -1); yp[idx] = y_tr[idx]
    a.append(SVC(C=10, gamma=gamma).fit(X_tr[idx], y_tr[idx]).score(X_te, y_te))
    b.append(np.mean(tsvm(X_tr, yp, C=10, gamma=gamma, Cu_final=1.0).predict(X_te) == 2 * y_te - 1))
    c.append(np.mean(tsvm(X_tr, yp, C=10, gamma=gamma, Cu_final=1.0, prop_pos=y_tr.mean()).predict(X_te) == 2 * y_te - 1))
print(f"gamma usado: {gamma:.4f} | proporção real de benignos no treino: {y_tr.mean():.2f} | proporção dos rotulados: 0.50")
print(f"SVM só rotulados          : {100 * np.mean(a):.1f}% ± {100 * np.std(a):.1f}")
for nome, v in [("TSVM (proporção dos rotulados)", b), ("TSVM (proporção real)", c)]:
    d = np.array(v) - np.array(a)
    print(f"{nome:26s}: {100 * np.mean(v):.1f}% ± {100 * np.std(v):.1f}  (ganho {100 * d.mean():+.1f} p.p.; melhor/pior {(d > 0).sum()}/{(d < 0).sum()})")
```
@@NOTE Com a proporção estimada nos rotulados (50% de benignos), o TSVM **piorou** 6,3 p.p. (melhor em 1 e pior em 19 dos 20 sorteios). Mas a proporção real de benignos no treino é de 63%. Com a proporção real, o TSVM **ganhou** 1,2 p.p. (melhor em 15 e pior em 5). A causa do prejuízo foi a proporção errada, e o resultado ilustra o risco do TSVM discutido no capítulo. Mesmo com a proporção correta, o ganho é pequeno: uma SVM com 10 rótulos já separa bem as classes neste conjunto.
@@EX 3 | Minimização de entropia no Breast Cancer | intermediário
No **Breast Cancer** (padronizado), com 10 rótulos (5 por classe) e 30 sorteios, use a regressão logística com minimização de entropia do capítulo e varie lambda em 0, 0,1, 0,3, 1 e 3 (com `l2 = 1`). Mostre a acurácia no teste e o ganho sobre lambda = 0.

*Pergunta:* qual lambda é melhor? Mudou o padrão visto nos dígitos 3 e 8?
@@SOL
```python
import numpy as np
from scipy.optimize import minimize
from sklearn.datasets import load_breast_cancer
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler

def sigmoide(z): return 1 / (1 + np.exp(-np.clip(z, -30, 30)))

def logistica_entropia(X, y_parcial, lam=1.0, l2=1.0, w0=None):
    """Regressão logística binária que minimiza: perda logística nos rotulados + lam * entropia das previsões nos sem rótulo + l2 * ||w||^2 (Grandvalet e Bengio, 2004)."""
    rot = y_parcial != -1; Xa = np.c_[X, np.ones(len(X))]
    def objetivo(w):
        z = Xa @ w; p = sigmoide(z)
        perda = -np.sum(y_parcial[rot] * np.log(p[rot] + 1e-12) + (1 - y_parcial[rot]) * np.log(1 - p[rot] + 1e-12))
        ent = -np.sum(p[~rot] * np.log(p[~rot] + 1e-12) + (1 - p[~rot]) * np.log(1 - p[~rot] + 1e-12))
        g = Xa[rot].T @ (p[rot] - y_parcial[rot]) + lam * Xa[~rot].T @ (-z[~rot] * p[~rot] * (1 - p[~rot])) + 2 * l2 * np.r_[w[:-1], 0]
        return perda + lam * ent + l2 * np.sum(w[:-1] ** 2), g
    w0 = np.zeros(Xa.shape[1]) if w0 is None else w0
    return minimize(objetivo, w0, jac=True, method="L-BFGS-B").x

X, y = load_breast_cancer(return_X_y=True)
X_tr, X_te, y_tr, y_te = train_test_split(X, y, test_size=0.3, stratify=y, random_state=0)
esc = StandardScaler().fit(X_tr); X_tr, X_te = esc.transform(X_tr), esc.transform(X_te)
res = {}
print("lambda | acurácia | ganho sobre lambda=0 (p.p.) | melhor/pior")
for lam in [0.0, 0.1, 0.3, 1.0, 3.0]:
    acc = []
    for s in range(30):
        r = np.random.RandomState(s); idx = np.r_[r.choice(np.where(y_tr == 0)[0], 5, replace=False), r.choice(np.where(y_tr == 1)[0], 5, replace=False)]
        yp = np.full(len(y_tr), -1); yp[idx] = y_tr[idx]
        w = logistica_entropia(X_tr, yp, lam=lam, l2=1.0)
        acc.append(np.mean((np.c_[X_te, np.ones(len(X_te))] @ w > 0) == y_te))
    res[lam] = np.array(acc); d = res[lam] - res[0.0]
    print(f"{lam:6.1f} | {100 * res[lam].mean():7.1f}% | {100 * d.mean():+26.1f} | {(d > 0).sum()}/{(d < 0).sum()}")
```
@@NOTE O melhor valor é lambda = 0,3 (91,6%, ganho de 1,3 p.p., melhor em 20 e pior em 8 dos 30 sorteios). Com lambda de 1 ou 3, o ganho some (0,0 e -0,2). O ganho é bem menor que nos dígitos 3 e 8 (de 87,6% para 95,7%), porque a regressão logística com 10 rótulos já alcança 90,3% no Breast Cancer, e há menos espaço para melhorar.

        perda = -np.sum(y_parcial[rot] * np.log(p[rot] + 1e-12) + (1 - y_parcial[rot]) * np.log(1 - p[rot] + 1e-12))
        ent = -np.sum(p[~rot] * np.log(p[~rot] + 1e-12) + (1 - p[~rot]) * np.log(1 - p[~rot] + 1e-12))
        g = Xa[rot].T @ (p[rot] - y_parcial[rot]) + lam * Xa[~rot].T @ (-z[~rot] * p[~rot] * (1 - p[~rot])) + 2 * l2 * np.r_[w[:-1], 0]
        return perda + lam * ent + l2 * np.sum(w[:-1] ** 2), g
    w0 = np.zeros(Xa.shape[1]) if w0 is None else w0
    return minimize(objetivo, w0, jac=True, method="L-BFGS-B").x

def painel(nome, X, y, n_rot, gamma=None, padronizar=True, sementes=20):
    X_tr, X_te, y_tr, y_te = train_test_split(X, y, test_size=0.4, stratify=y, random_state=0)
    if padronizar:
        esc = StandardScaler().fit(X_tr); X_tr, X_te = esc.transform(X_tr), esc.transform(X_te)
    g = gamma if gamma is not None else 1 / (X_tr.shape[1] * X_tr.var())
    base, ts, en, dentro = [], [], [], []
    for s in range(sementes):
        r = np.random.RandomState(s)
        idx = np.concatenate([r.choice(np.where(y_tr == c)[0], n_rot // 2, replace=False) for c in (0, 1)])
        yp = np.full(len(y_tr), -1); yp[idx] = y_tr[idx]
        m = SVC(C=10, gamma=g).fit(X_tr[idx], y_tr[idx]); base.append(m.score(X_te, y_te))
        sem = yp == -1; dentro.append(np.mean(np.abs(m.decision_function(X_tr[sem])) < 1))
        ts.append(np.mean(tsvm(X_tr, yp, C=10, gamma=g, Cu_final=1.0).predict(X_te) == 2 * y_te - 1))
        w = logistica_entropia(X_tr, yp, lam=0.3, l2=1.0); en.append(np.mean((np.c_[X_te, np.ones(len(X_te))] @ w > 0) == y_te))
    base, ts, en = map(np.array, (base, ts, en))
    return {"base": nome, "SVM (%)": round(100 * base.mean(), 1), "TSVM (%)": round(100 * ts.mean(), 1), "ganho TSVM": round(100 * (ts - base).mean(), 1),
            "entropia (%)": round(100 * en.mean(), 1), "ganho entropia": round(100 * (en - base).mean(), 1), "sem rótulo na margem (%)": round(100 * np.mean(dentro), 0)}

Xl, yl = make_moons(400, noise=0.1, random_state=0)
